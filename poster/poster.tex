%% ── Section-header label ────────────────────────────────────

\newcommand{\secheader}[1]{%
  \vspace*{0.55cm}%
  \noindent{%
    \setlength{\fboxsep}{5pt}%
    \colorbox{lightheader}{%
      \textbf{\fontsize{28}{34}\selectfont\strut #1}%
    }%
  }\par\vspace{0.30cm}%
}

%% ── Figure / image placeholder ──────────────────────────────
\newcommand{\figplaceholder}[2][8.5cm]{%

        \centering\color{posterblue}\textit{#2}%
      }%
    }%
  \end{center}%
}

  \par\vspace{0.1cm}%
