%% ── Background ───────────────────────────────────────────────
\secheader{Background: Link Prediction}

A \textbf{knowledge graph} (KG) is a directed multi-relational graph
$\mathcal{G}=(\mathcal{E},\mathcal{R},\mathcal{T})$ where each triple
$(h,r,t)$ encodes a binary fact.
\textbf{Link prediction} asks: given $(h,r,?)$, rank all
$|\mathcal{E}|$ candidate entities by plausibility.
The standard metric is \textbf{MRR} (mean reciprocal rank of the
correct entity in the filtered ranking).

\vspace{0.25cm}

\vspace{0.1cm}
Training requires \emph{negative} triples (false facts).
\textbf{Random} Bernoulli corruption yields easy negatives after a few
epochs --- gradient signal collapses.
\textbf{Hard negatives} (triples the model still scores highly) carry
more information but risk instability if used alone.

%% ── Models & Setup ───────────────────────────────────────────
\secheader{Models \& Training Setup}

\textbf{TransE:}\ $f(h,r,t)=-\|\mathbf{h}+\mathbf{r}-\mathbf{t}\|_2$
\hfill(translation in $\mathbb{R}^d$)

\vspace{0.15cm}
\textbf{RotatE:}\ $f(h,r,t)=-\|\mathbf{h}\circ\mathbf{r}-\mathbf{t}\|$,\
$|r_k|{=}1$ \hfill(rotation in $\mathbb{C}^d$)

\vspace{0.15cm}
\textbf{NSSALoss} (self-adversarial):
\[
\mathcal{L}=-\log\sigma(\gamma-f_+)
  -\sum_i p_i\log\sigma(f_i'-\gamma)
\]
where $p_i=e^{\alpha f_i'}/\sum_j e^{\alpha f_j'}$ (margin $\gamma{=}9.0$,
temp.\ $\alpha{=}1.0$).
Hard negatives get higher weight $p_i$, amplifying gradient signal.

\vspace{0.15cm}
{\small \textbf{All runs:} $d=128$, batch 1024, Adam lr$=10^{-3}$,
50 epochs (patience 10), seed 42, $n=64$ pool, $k=8$ negs/positive.}

%% ── Pipeline ─────────────────────────────────────────────────
\secheader{Two-Stage Negative Sampling}

\vspace{0.12cm}
\begin{center}
\begin{tikzpicture}[
  node distance=0.38cm and 0.45cm,
  font=\small, >=Latex,
  box/.style={draw, align=center, inner sep=3pt, rounded corners=3pt,
              minimum width=3.0cm, minimum height=0.72cm,
              fill=posterblue!9, draw=posterblue!50},
  stg/.style={font=\scriptsize\bfseries, text=posterblue},
]
  \node[box] (pos)    {Positive\\$(h,r,t)$};
  \node[box, right=0.45cm of pos]  (bern)  {Bernoulli\\side};
  \node[box, right=0.45cm of bern] (pool)  {Pool $n{=}64$\\candidates};
  \node[box, below=0.45cm of pool] (score) {Score with\\$f_\theta$};
  \node[box, left=0.45cm of score] (sel)   {Select $k{=}8$\\strategy};
  \node[box, left=0.45cm of sel]   (loss)  {NSSA\\Loss};
  \path[->]
    (pos) edge (bern) (bern) edge (pool) (pool) edge (score)
    (score) edge (sel) (sel) edge (loss);
  \node[stg, above=0.05cm of pos]  {Stage 1};
  \node[stg, above=0.05cm of pool] {Stage 1};
  \node[stg, below=0.05cm of score]{Stage 2};
  \node[stg, below=0.05cm of loss] {Stage 2};
\end{tikzpicture}
\end{center}

\vspace{0.1cm}
\textbf{Stage~1:} Bernoulli corruption draws $n{=}64$ candidates,
filtering known-true triples.\\[3pt]
\textbf{Stage~2:} selects $k{=}8$ via \textbf{Random} (control),
\textbf{Hard} (top-$k$ by $f_\theta$), or \textbf{Mixed $p$}
($p\in\{0.3,0.5,0.7\}$).
All strategies use the same pool --- \textbf{zero extra compute cost}.

\colsep

%% ── Conclusions ──────────────────────────────────────────────
\secheader{Conclusions}

\begin{itemize}\setlength{\itemsep}{4pt}
  \item \textbf{Hard negatives $+$10.4\% MRR} over random sampling,
    consistent across all metrics and reproducible with seed 42.
  \item \textbf{Mixed 30\% recovers 95\% of the gain} at zero
    extra compute; recommended default for practitioners.
  \item \textbf{Gains concentrate on rare relations} ($+$3.6 pp)
    and high-degree tail entities, where hard negatives maximise
    gradient information density.
  \item \textbf{Future:} curriculum scheduling of $p$; apply to
    RESCAL and TuckER; test on Wikidata-5M scale.
\end{itemize}
